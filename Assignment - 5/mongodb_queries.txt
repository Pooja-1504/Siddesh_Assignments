// SOCIAL MEDIA PLATFORM DATABASE

use SocialMediaDB


// COLLECTIONS 

// USERS
db.Users.insertMany([
  { name: "Pooja", city: "Pune" },
  { name: "Riya", city: "Mumbai" },
  { name: "Amit", city: "Delhi" }
])

// FOLLOWERS
db.Followers.insertMany([
  { user: "Pooja", follower: "Riya" },
  { user: "Pooja", follower: "Amit" },
  { user: "Riya", follower: "Pooja" }
])

// POSTS (Nested Likes & Comments)
db.Posts.insertMany([
{
  user: "Pooja",
  content: "Hello MongoDB",
  likes: ["Riya","Amit"],
  comments: [
    { user: "Riya", text: "Nice post!" }
  ],
  date: new Date()
},
{
  user: "Riya",
  content: "Good morning",
  likes: ["Pooja"],
  comments: [],
  date: new Date()
}
])

// COMMENTS 
db.Comments.insertMany([
  { postUser:"Pooja", commentUser:"Riya", text:"Nice post!" }
])

// LIKES 
db.Likes.insertMany([
  { postUser:"Pooja", likedBy:"Amit" }
])


// Users with MOST followers
db.Followers.aggregate([
  { $group:{ _id:"$user", totalFollowers:{ $sum:1 }}},
  { $sort:{ totalFollowers:-1 }}
])


// Posts with highest likes/comments
db.Posts.aggregate([
{
  $project:{
    content:1,
    likesCount:{ $size:"$likes" },
    commentsCount:{ $size:"$comments" }
  }
},
{ $sort:{ likesCount:-1 }}
])


// Aggregate user activity over time
db.Posts.aggregate([
{
  $group:{
    _id:{ month:{ $month:"$date" }},
    totalPosts:{ $sum:1 }
  }
}
])


// 4. Nested documents already shown in Posts collection
db.Posts.find().pretty()


// ONLINE STORE / E-COMMERCE DATABASE

use OnlineStoreDB


// COLLECTIONS 

// PRODUCTS
db.Products.insertMany([
  { name:"Phone", category:"Electronics", price:20000 },
  { name:"Laptop", category:"Electronics", price:60000 },
  { name:"Shoes", category:"Fashion", price:3000 }
])

// CUSTOMERS
db.Customers.insertMany([
  { name:"Pooja" },
  { name:"Riya" }
])

// ORDERS (Embedded order items)
db.Orders.insertMany([
{
  customer:"Pooja",
  items:[
    { product:"Phone", qty:2 },
    { product:"Shoes", qty:1 }
  ],
  total:43000,
  date:new Date()
},
{
  customer:"Riya",
  items:[
    { product:"Laptop", qty:1 }
  ],
  total:60000,
  date:new Date()
}
])

// REVIEWS
db.Reviews.insertMany([
  { product:"Phone", rating:5, review:"Very good" },
  { product:"Laptop", rating:4, review:"Nice laptop" }
])

// CART
db.Cart.insertOne({
  customer:"Pooja",
  items:[
    { product:"Laptop", qty:1 }
  ]
})


// TOP-SELLING PRODUCTS PER CATEGORY
db.Orders.aggregate([
  { $unwind:"$items" },
  {
    $lookup:{
      from:"Products",
      localField:"items.product",
      foreignField:"name",
      as:"productInfo"
    }
  },
  { $unwind:"$productInfo" },
  {
    $group:{
      _id:{
        category:"$productInfo.category",
        product:"$items.product"
      },
      totalSold:{ $sum:"$items.qty" }
    }
  },
  { $sort:{ totalSold:-1 }}
])


// TRACK CUSTOMER PURCHASE PATTERNS
db.Orders.aggregate([
{
  $group:{
    _id:"$customer",
    totalOrders:{ $sum:1 },
    totalSpent:{ $sum:"$total" }
  }
}
])


// AGGREGATE REVENUE MONTHLY & YEARLY
db.Orders.aggregate([
{
  $group:{
    _id:{
      year:{ $year:"$date" },
      month:{ $month:"$date" }
    },
    revenue:{ $sum:"$total" }
  }
}
])

// Embedded documents already shown in Orders (items array)
db.Orders.find().pretty()
